4. Xử lý trên Google Earth Engine
Bước 1: Lấy vùng nghiên cứu
var region = ee.FeatureCollection("users/your_account/kon_tum_boundary");Map.centerObject(region, 8);
Bước 2: Lọc ảnh Sentinel-2
var s2 = ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")  .filterBounds(region)  .filterDate('2026-03-01', '2026-05-24')  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))  .median()  .clip(region);
Bước 3: Tính NDVI, NDMI, NBR
var ndvi = s2.normalizedDifference(['B8', 'B4']).rename('NDVI');var ndmi = s2.normalizedDifference(['B8', 'B11']).rename('NDMI');var nbr  = s2.normalizedDifference(['B8', 'B12']).rename('NBR');
Bước 4: Lấy nhiệt độ bề mặt MODIS
var lst = ee.ImageCollection("MODIS/061/MOD11A1")  .filterBounds(region)  .filterDate('2026-05-01', '2026-05-24')  .select('LST_Day_1km')  .mean()  .multiply(0.02)  .subtract(273.15)  .rename('LST')  .clip(region);
Bước 5: Tạo điểm nguy cơ cháy
var lstScore = lst.gt(35).multiply(1);var dryScore = ndmi.lt(0.2).multiply(1);var vegScore = ndvi.gt(0.3).multiply(1);var nbrScore = nbr.lt(0.3).multiply(1);var fireRisk = lstScore.multiply(0.35)  .add(dryScore.multiply(0.30))  .add(vegScore.multiply(0.20))  .add(nbrScore.multiply(0.15))  .rename('FireRisk');
Bước 6: Phân cấp cảnh báo
var riskClass = fireRisk.expression(  "(b('FireRisk') < 0.2) ? 1" +  ": (b('FireRisk') < 0.4) ? 2" +  ": (b('FireRisk') < 0.6) ? 3" +  ": (b('FireRisk') < 0.8) ? 4" +  ": 5").rename('risk_level');
Bước 7: Hiển thị bản đồ cảnh báo
Map.addLayer(riskClass, {  min: 1,  max: 5,  palette: ['00cc44', 'ffff00', 'ff9900', 'ff0000', '800000']}, 'Cảnh báo nguy cơ cháy rừng');
5. Tích hợp NASA FIRMS để xác nhận điểm cháy
NASA FIRMS nên dùng như lớp điểm cháy thực tế, còn GEE dùng để tính nguy cơ cháy.
Nguy cơ cháy cao + Có điểm FIRMS gần đó→ Cảnh báo ưu tiên cao
Dữ liệu VIIRS Active Fire có độ phân giải khoảng 375 m, phù hợp cho quản lý cháy rừng gần thời gian thực. 
6. Lưu kết quả về PostGIS
Bảng đề xuất:
CREATE TABLE forest_fire_warning (  id SERIAL PRIMARY KEY,  risk_level INTEGER,  risk_score NUMERIC,  source VARCHAR(50),  province VARCHAR(100),  district VARCHAR(100),  commune VARCHAR(100),  forest_type VARCHAR(100),  warning_time TIMESTAMP,  geom GEOMETRY(Polygon, 4326));
Bảng điểm cháy FIRMS:
CREATE TABLE active_fire_point (  id SERIAL PRIMARY KEY,  latitude NUMERIC,  longitude NUMERIC,  brightness NUMERIC,  confidence VARCHAR(50),  satellite VARCHAR(50),  acq_date DATE,  acq_time VARCHAR(10),  geom GEOMETRY(Point, 4326));
